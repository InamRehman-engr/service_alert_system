# INA219

``` c
float volt = 0;
float current_mA = 0;
float voltShunt = 0;

INA219_Init(0x40); // 0x40 is I2C address of INA219

// this sets the calibration of chip
// VBUS_MAX to 16V 
// RSHUNT = 100mOhm
int INAreturn = INA219_setCalibration_16V_1A_100mOhm();
if (INAreturn == 0)
{
    // Get bus voltage 
    INA219_getBusVoltage_V(&volt);

    // Get Current 
    INA219_getCurrent_mA(&current_mA);

    // this is voltage across shunt
    INA219_getShuntVoltage_mV(&voltShunt);

    // puts the chip in sleep
    INA219_setSleepMode();

}

```
